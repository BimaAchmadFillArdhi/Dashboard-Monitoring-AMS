{# Reusable Badge and StatusPill Macros #} {% macro badge(text="", tone="neutral", size="sm", extra_classes="") %} {%- if size == "xs" -%} {%- set size_cls = "text-[9.5px] px-1.5 py-0.5 rounded" -%} {%- else -%} {%- set size_cls = "text-[10.5px] px-2 py-0.5 rounded" -%} {%- endif -%} {%- if tone == "success" or tone == "healthy" -%} {%- set tone_cls = "badge-success border" -%} {%- elif tone == "warning" or tone == "perhatian" -%} {%- set tone_cls = "badge-warning border" -%} {%- elif tone == "danger" or tone == "bermasalah" or tone == "error" -%} {%- set tone_cls = "badge-danger border" -%} {%- elif tone == "info" or tone == "primary" -%} {%- set tone_cls = "badge-info border" -%} {%- else -%} {%- set tone_cls = "badge-neutral border" -%} {%- endif -%} {{ text }} {% endmacro %} {% macro status_pill(status="", custom_label=None, size="sm") %} {%- set s = status | lower -%} {%- if s in ['healthy', 'sehat', 'broadcasting', 'live'] -%} {%- set label = custom_label or 'Sehat' -%} {%- set dot_cls = "badge-dot-success" -%} {%- set pill_cls = "badge-success" -%} {%- elif s in ['warning', 'perhatian', 'warn'] -%} {%- set label = custom_label or 'Perhatian' -%} {%- set dot_cls = "badge-dot-warning" -%} {%- set pill_cls = "badge-warning" -%} {%- elif s in ['danger', 'bermasalah', 'error'] -%} {%- set label = custom_label or 'Bermasalah' -%} {%- set dot_cls = "badge-dot-danger" -%} {%- set pill_cls = "badge-danger" -%} {%- elif s in ['info'] -%} {%- set label = custom_label or 'Info' -%} {%- set dot_cls = "badge-dot-info" -%} {%- set pill_cls = "badge-info" -%} {%- else -%} {# offline, finished, created, debug, unknown #} {%- set label = custom_label or (s | capitalize if s else 'Offline') -%} {%- set dot_cls = "badge-dot-neutral" -%} {%- set pill_cls = "badge-neutral" -%} {%- endif -%} {{ label }} {% endmacro %}
